> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Spotle-style daily video-game guessing game as a static page, plus a one-shot Node script that generates its ~3,000-game dataset from IGDB + Steam.

**Architecture:** Static site (`index.html` + `style.css` + `game.js`) driven by a pure, DOM-free `logic.js` that is also unit-tested under Node. Data is a generated `data.js` (`const GAMES = [...]`) produced by `scripts/build-data.js`, which pulls IGDB candidates, applies exclusion rules from `scripts/config.js`, enriches with SteamSpy tags and Steam store Metacritic scores, and caches every API response in `scripts/.cache/`.

**Tech Stack:** Vanilla JS (browser, no framework, no bundler), Node 24 (`node:test`, built-in `fetch`), zero npm dependencies.

**Spec:** `docs/superpowers/specs/2026-10-13-gtgdle-design.md`

## Global Constraints

- **Do NOT commit.** Justin commits only when he says so. Each task ends with a checkpoint (`git status -- .` shows only the expected files), not a commit.
- Zero npm dependencies. Tests run with `node --test`; scripts use Node's built-in `fetch`.
- No runtime network calls from the game except the Google Fonts stylesheet. All game data comes from `data.js`.
- `MAX_GUESSES = 10`. Year yellow = |Δ| ≤ 3. Metacritic yellow = |Δ| ≤ 5. Daily day #1 = `2026-10-13`.
- Pool: top 3,000 survivors by IGDB `total_rating_count`, from ~5,000 candidates.
- Primary tags: Steam top 5 (else IGDB genres + themes). Secondary: Steam ranks 6–20 (else IGDB keywords, first 15).
- Metacritic: Steam store score, else IGDB `aggregated_rating` rounded, else `null`.
- Twitch credentials come only from env vars `TWITCH_CLIENT_ID` / `TWITCH_CLIENT_SECRET`; never written to any file.
- All game-data strings are rendered with `textContent`, never `innerHTML`.
- Display name "GTGdle" is a placeholder; keep it in one constant (`GAME_TITLE` in `logic.js`) plus the `<title>`/`<h1>` in `index.html`.

## Review Focus

1. **Accented/punctuated names typed plainly** ("pokemon", "okami", "assassins creed") must find "Pokémon …", "Ōkami", "Assassin's Creed". Pinned in Task 1 (`normalize`) and Task 3 (`searchGames`).
2. **Distinct games sharing a name** (Doom 1993 vs Doom 2016) must both survive the build and be guessable, distinguished by year. Pinned in Task 4 (`selectGames` dedupes on name **+ year**, a deliberate refinement of the spec's "normalized name" rule).
3. **Corrupt or stale localStorage** (bad JSON, ids no longer in data, >10 guesses) must yield a fresh board, never a crash. Pinned in Task 3 (`parseSavedState`).
4. **Sparse game records** (empty tag/platform/developer lists, null franchise, null metacritic) must render gray "No match" / "—" / "N/A" without throwing. Pinned in Task 2.
5. **Re-guessing an already-guessed game** must be impossible: it is never offered in suggestions. Pinned in Task 3 (`searchGames` `excludeIds`).

```js
'use strict';

// ═══════════════════════════════════════════════════════════════════════════
//  CONSTANTS
// ═══════════════════════════════════════════════════════════════════════════
const GAME_TITLE = 'GTGdle';
const MAX_GUESSES = 10;
const DAY_EPOCH = '2026-10-13';

// ═══════════════════════════════════════════════════════════════════════════
//  TEXT
// ═══════════════════════════════════════════════════════════════════════════
function normalize(str) {
  return String(str ?? '')
    .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .replace(/&/g, ' and ')
    .replace(/['\u2019]/g, '')
    .replace(/[^a-z0-9]+/g, ' ')
    .trim();
}

// ═══════════════════════════════════════════════════════════════════════════
//  DATES + SEEDING
// ═══════════════════════════════════════════════════════════════════════════
function todayStr(d = new Date()) {
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
}
function dateToSeed(s) { return parseInt(s.replace(/-/g, ''), 10); }
function dayNumber(s) {
  return Math.round((Date.parse(`${s}T00:00:00Z`) - Date.parse(`${DAY_EPOCH}T00:00:00Z`)) / 86400000) + 1;
}

function mulberry32(seed) {
  let s = seed | 0;
  return () => {
    s = (s + 0x6D2B79F5) | 0;
    let z = Math.imul(s ^ (s >>> 15), 1 | s);
    z = (z + Math.imul(z ^ (z >>> 7), 61 | z)) ^ z;
    return ((z ^ (z >>> 14)) >>> 0) / 4294967296;
  };
}

function pickDaily(games, dateStr) {
  return games[Math.floor(mulberry32(dateToSeed(dateStr))() * games.length)];
}
function pickRandom(games, rand = Math.random) {
  return games[Math.floor(rand() * games.length)];
}

// ═══════════════════════════════════════════════════════════════════════════
//  EXPORTS (Node only — in the browser these are plain globals)
// ═══════════════════════════════════════════════════════════════════════════
if (typeof module !== 'undefined') {
  module.exports = {
    GAME_TITLE, MAX_GUESSES, DAY_EPOCH,
    normalize, todayStr, dateToSeed, dayNumber, mulberry32, pickDaily, pickRandom,
  };
}
```

```js
// ═══════════════════════════════════════════════════════════════════════════
//  COMPARISONS — each returns { status, display, ... }
// ═══════════════════════════════════════════════════════════════════════════
// Items of `guessList` whose normalized form appears in `answerList`, in guess order.
function intersect(guessList, answerList) {
  const answerSet = new Set(answerList.map(normalize));
  return guessList.filter(x => answerSet.has(normalize(x)));
}

function sameSet(x, y) {
  const a = new Set(x.map(normalize)), b = new Set(y.map(normalize));
  return a.size === b.size && [...a].every(v => b.has(v));
}

function summarize(list) {
  if (list.length <= TAG_DISPLAY_LIMIT) return list.join(', ');
  return `${list.slice(0, TAG_DISPLAY_LIMIT).join(', ')} +${list.length - TAG_DISPLAY_LIMIT}`;
}

function compareNumber(guessVal, answerVal, delta) {
  if (guessVal == null || answerVal == null) return { status: 'gray', display: 'N/A', arrow: null };
  if (guessVal === answerVal) return { status: 'green', display: String(guessVal), arrow: null };
  return {
    status: Math.abs(guessVal - answerVal) <= delta ? 'yellow' : 'gray',
    display: String(guessVal),
    arrow: answerVal > guessVal ? 'up' : 'down',
  };
}

function compareYear(g, a) { return compareNumber(g.year, a.year, YEAR_YELLOW_DELTA); }
function compareMetacritic(g, a) { return compareNumber(g.metacritic, a.metacritic, SCORE_YELLOW_DELTA); }

function compareTags(g, a) {
  const primary = intersect(g.primaryTags, a.primaryTags);
  if (primary.length) return { status: 'green', display: summarize(primary), matched: primary };
  const any = intersect([...g.primaryTags, ...g.secondaryTags], [...a.primaryTags, ...a.secondaryTags]);
  if (any.length) return { status: 'yellow', display: summarize(any), matched: any };
  return { status: 'gray', display: 'No match', matched: [] };
}

function comparePlatforms(g, a) {
  const shared = intersect(g.platforms, a.platforms);
  if (shared.length && sameSet(g.platforms, a.platforms)) return { status: 'green', display: g.platforms.join(', ') };
  if (shared.length) return { status: 'yellow', display: shared.join(', ') };
  return { status: 'gray', display: g.platforms.join(', ') || '—' };
}

function compareDevelopers(g, a) {
  const shared = intersect(g.developers, a.developers);
  return { status: shared.length ? 'green' : 'gray', display: g.developers.join(', ') || '—' };
}

function compareFranchise(g, a) {
  const match = g.franchise && a.franchise && normalize(g.franchise) === normalize(a.franchise);
  return { status: match ? 'green' : 'gray', display: g.franchise || '—' };
}

function compareGuess(g, a) {
  return {
    year: compareYear(g, a),
    tags: compareTags(g, a),
    platforms: comparePlatforms(g, a),
    developers: compareDevelopers(g, a),
    franchise: compareFranchise(g, a),
    metacritic: compareMetacritic(g, a),
  };
}
```

```js
'use strict';
// Pure transforms from raw IGDB/Steam responses to GAMES entries. No I/O.
const { normalize } = require('../logic.js');

function startsWithWords(text, prefix) {
  const t = normalize(text), p = normalize(prefix);
  return p.length > 0 && (t === p || t.startsWith(`${p} `));
}

function containsWords(text, phrase) {
  const p = normalize(phrase);
  return p.length > 0 && ` ${normalize(text)} `.includes(` ${p} `);
}

function releaseYear(raw) {
  return new Date(raw.first_release_date * 1000).getUTCFullYear();
}

function franchiseNames(raw) {
  return [...(raw.franchises || []), ...(raw.collections || []), ...(raw.collection ? [raw.collection] : [])]
    .map(f => f && f.name)
    .filter(Boolean);
}

// ── Exclusion ────────────────────────────────────────────────────────────────
function classify(raw, cfg) {
  if (cfg.forceInclude.includes(raw.id)) return null;
  if (cfg.forceExclude.includes(raw.id)) return 'manual forceExclude';
  const names = [raw.name, ...franchiseNames(raw)];
  const blocked = cfg.blockedFranchises.find(b => names.some(n => startsWithWords(n, b)));
  if (blocked) return `blocked franchise: ${blocked}`;
  const genres = (raw.genres || []).map(g => g.name);
  if (genres.includes('Sport') && !cfg.sportAllowlist.some(a => containsWords(raw.name, a))) return 'sport genre';
  return null;
}

function selectGames(raws, cfg, limit = 3000) {
  const kept = [], excluded = [];
  const seen = new Map(); // "normalized name|year" → kept id
  for (const r of raws) {
    if (kept.length >= limit) break;
    const reason = classify(r, cfg);
    if (reason) { excluded.push({ id: r.id, name: r.name, reason }); continue; }
    const key = `${normalize(r.name)}|${releaseYear(r)}`;
    if (seen.has(key)) { excluded.push({ id: r.id, name: r.name, reason: `duplicate of ${seen.get(key)}` }); continue; }
    seen.set(key, r.id);
    kept.push(r);
  }
  return { kept, excluded };
}

// ── Tags ─────────────────────────────────────────────────────────────────────
function titleCase(s) {
  return s.replace(/\b[a-z]/g, c => c.toUpperCase());
}

function cleanTag(tag, cfg) {
  const trimmed = String(tag ?? '').trim();
  if (!trimmed) return null;
  const alias = cfg.tagAliases[trimmed.toLowerCase()];
  const name = alias ?? (trimmed === trimmed.toLowerCase() ? titleCase(trimmed) : trimmed);
  return cfg.tagBlocklist.some(b => normalize(b) === normalize(name)) ? null : name;
}

function cleanTagList(tags, cfg) {
  const out = [], seen = new Set();
  for (const t of tags) {
    const name = cleanTag(t, cfg);
    if (!name || seen.has(normalize(name))) continue;
    seen.add(normalize(name));
    out.push(name);
  }
  return out;
}

function deriveTags(raw, steamTags, cfg) {
  if (steamTags && steamTags.length) {
    const tags = cleanTagList(steamTags, cfg);
    return { primaryTags: tags.slice(0, 5), secondaryTags: tags.slice(5, 20) };
  }
  const names = list => (list || []).map(x => x.name);
  const base = cleanTagList([...names(raw.genres), ...names(raw.themes)], cfg);
  const primaryTags = base.slice(0, 5);
  const all = cleanTagList([...primaryTags, ...base.slice(5), ...names(raw.keywords)], cfg);
  return { primaryTags, secondaryTags: all.slice(primaryTags.length, primaryTags.length + 15) };
}

// ── Other fields ─────────────────────────────────────────────────────────────
function normalizePlatforms(raw, cfg, onUnmapped = () => {}) {
  const out = [];
  for (const p of raw.platforms || []) {
    const mapped = cfg.platformMap[p.name];
    if (mapped === undefined) { onUnmapped(p.name); continue; }
    if (mapped && !out.includes(mapped)) out.push(mapped);
  }
  return out;
}

function developers(raw) {
  const out = [];
  for (const c of raw.involved_companies || []) {
    const name = c.developer && c.company && c.company.name;
    if (name && !out.includes(name)) out.push(name);
  }
  return out;
}

function buildEntry(raw, steam, cfg, onUnmapped) {
  const { primaryTags, secondaryTags } = deriveTags(raw, steam.tags, cfg);
  const igdbScore = raw.aggregated_rating != null ? Math.round(raw.aggregated_rating) : null;
  return {
    id: raw.id,
    name: raw.name,
    year: releaseYear(raw),
    primaryTags,
    secondaryTags,
    platforms: normalizePlatforms(raw, cfg, onUnmapped),
    developers: developers(raw),
    franchise: franchiseNames(raw)[0] ?? null,
    metacritic: steam.metacritic ?? igdbScore,
  };
}

// ── Steam response parsing ───────────────────────────────────────────────────
const STEAM_SOURCE = 1; // IGDB external source id for Steam (both `category` and `external_game_source`)

function steamAppId(raw) {
  const ext = (raw.external_games || []).find(e =>
    (e.external_game_source === STEAM_SOURCE || e.category === STEAM_SOURCE) && /^\d+$/.test(String(e.uid)));
  return ext ? Number(ext.uid) : null;
}

function steamSpyTags(json) {
  return Object.entries((json && json.tags) || {})
    .sort((a, b) => b[1] - a[1])
    .map(([name]) => name);
}

function steamMetacritic(json, appId) {
  const entry = json && json[appId];
  return entry && entry.success && entry.data?.metacritic?.score != null ? entry.data.metacritic.score : null;
}

module.exports = {
  startsWithWords, containsWords, releaseYear, classify, selectGames,
  cleanTagList, deriveTags, normalizePlatforms, buildEntry,
  steamAppId, steamSpyTags, steamMetacritic,
};
```

```js
'use strict';
// Network + disk cache for the build. Every fetch takes an injectable fetchFn for tests.
const fs = require('node:fs');
const path = require('node:path');
const { steamSpyTags, steamMetacritic } = require('./transform.js');

const sleep = ms => (ms > 0 ? new Promise(r => setTimeout(r, ms)) : Promise.resolve());

function createCache(dir) {
  fs.mkdirSync(dir, { recursive: true });
  const file = key => path.join(dir, `${key}.json`);
  return {
    get(key) { return fs.existsSync(file(key)) ? JSON.parse(fs.readFileSync(file(key), 'utf8')) : undefined; },
    set(key, value) { fs.writeFileSync(file(key), JSON.stringify(value)); },
  };
}

async function getTwitchToken(clientId, clientSecret, fetchFn = fetch) {
  const url = 'https://id.twitch.tv/oauth2/token'
    + `?client_id=${encodeURIComponent(clientId)}&client_secret=${encodeURIComponent(clientSecret)}&grant_type=client_credentials`;
  const res = await fetchFn(url, { method: 'POST' });
  if (!res.ok) throw new Error(`Twitch auth failed: HTTP ${res.status} ${await res.text()}`);
  return (await res.json()).access_token;
}

const IGDB_PAGE = 500;
const IGDB_FIELDS = [
  'name', 'first_release_date', 'total_rating_count', 'aggregated_rating',
  'genres.name', 'themes.name', 'keywords.name', 'platforms.name',
  'involved_companies.developer', 'involved_companies.company.name',
  'franchises.name', 'collections.name',
  'external_games.uid', 'external_games.category', 'external_games.external_game_source',
].join(',');

// game_type: 0 main, 4 standalone expansion, 8 remake, 9 remaster, 10 expanded game.
// If IGDB rejects `game_type`, swap to the legacy `category = (0,4,8,9,10)` — same ids.
function igdbQuery(offset) {
  return `fields ${IGDB_FIELDS}; `
    + 'where game_type = (0,4,8,9,10) & first_release_date != null & total_rating_count != null & version_parent = null; '
    + `sort total_rating_count desc; limit ${IGDB_PAGE}; offset ${offset};`;
}

async function fetchIgdbCandidates({ clientId, token, cache, total = 5000, fetchFn = fetch, delayMs = 300, log = console.log }) {
  const all = [];
  for (let offset = 0; offset < total; offset += IGDB_PAGE) {
    const key = `igdb-${offset}`;
    let page = cache.get(key);
    if (page === undefined) {
      const res = await fetchFn('https://api.igdb.com/v4/games', {
        method: 'POST',
        headers: { 'Client-ID': clientId, Authorization: `Bearer ${token}`, Accept: 'application/json' },
        body: igdbQuery(offset),
      });
      if (!res.ok) throw new Error(`IGDB request failed at offset ${offset}: HTTP ${res.status} ${await res.text()}`);
      page = await res.json();
      cache.set(key, page);
      await sleep(delayMs);
    }
    all.push(...page);
    log(`IGDB: ${all.length} candidates`);
    if (page.length < IGDB_PAGE) break;
  }
  return all;
}

async function fetchJsonWithRetry(url, fetchFn, retryDelayMs) {
  for (let attempt = 0; attempt < 2; attempt++) {
    try {
      const res = await fetchFn(url);
      if (res.ok) return await res.json();
    } catch { /* network error — retry */ }
    if (attempt === 0) await sleep(retryDelayMs);
  }
  return null;
}

async function cachedJson(cache, key, url, fetchFn, delayMs, retryDelayMs) {
  const hit = cache.get(key);
  if (hit !== undefined) return hit;
  const json = await fetchJsonWithRetry(url, fetchFn, retryDelayMs);
  if (json) cache.set(key, json); // failures are not cached so a rerun retries them
  await sleep(delayMs);
  return json;
}

async function fetchSteamData(appId, { cache, fetchFn = fetch, spyDelayMs = 1000, storeDelayMs = 1500, retryDelayMs = 5000 }) {
  const spy = await cachedJson(cache, `steamspy-${appId}`,
    `https://steamspy.com/api.php?request=appdetails&appid=${appId}`, fetchFn, spyDelayMs, retryDelayMs);
  const store = await cachedJson(cache, `steamstore-${appId}`,
    `https://store.steampowered.com/api/appdetails?appids=${appId}&filters=metacritic`, fetchFn, storeDelayMs, retryDelayMs);
  return {
    tags: spy ? steamSpyTags(spy) : null,
    metacritic: store ? steamMetacritic(store, appId) : null,
    failed: !spy || !store,
  };
}

module.exports = { createCache, getTwitchToken, fetchIgdbCandidates, fetchSteamData };
```

```js
#!/usr/bin/env node
'use strict';
// Generates ../data.js and ../excluded.txt from IGDB + SteamSpy + Steam store.
//
// Usage (PowerShell):
//   $env:TWITCH_CLIENT_ID = '...'; $env:TWITCH_CLIENT_SECRET = '...'; npm run build-data
//
// Get credentials by registering an app at https://dev.twitch.tv/console/apps (category: Application
// Integration, OAuth redirect: http://localhost). Responses are cached in scripts/.cache/; delete that
// folder to force a full refresh. A cold run takes roughly an hour (Steam rate limits).
const fs = require('node:fs');
const path = require('node:path');
const config = require('./config.js');
const { selectGames, buildEntry, steamAppId } = require('./transform.js');
const { createCache, getTwitchToken, fetchIgdbCandidates, fetchSteamData } = require('./sources.js');

const ROOT = path.join(__dirname, '..');
const POOL_SIZE = 3000;
const CANDIDATES = 5000;

function renderDataJs(games, dateStr) {
  return `// GENERATED by scripts/build-data.js on ${dateStr} — do not edit by hand.\n`
    + `const GAMES = [\n${games.map(g => JSON.stringify(g)).join(',\n')}\n];\n`
    + `if (typeof module !== 'undefined') module.exports = GAMES;\n`;
}

function renderExcluded(excluded) {
  return '# id\tname\treason  (move games back with forceInclude in scripts/config.js)\n'
    + excluded.map(e => `${e.id}\t${e.name}\t${e.reason}\n`).join('');
}

function pct(n, total) { return total ? `${Math.round((n / total) * 100)}%` : '0%'; }

async function main() {
  const clientId = process.env.TWITCH_CLIENT_ID;
  const clientSecret = process.env.TWITCH_CLIENT_SECRET;
  if (!clientId || !clientSecret) {
    console.error('Missing TWITCH_CLIENT_ID / TWITCH_CLIENT_SECRET environment variables. See header of scripts/build-data.js.');
    process.exit(1);
  }

  const cache = createCache(path.join(__dirname, '.cache'));
  const token = await getTwitchToken(clientId, clientSecret);
  const raws = await fetchIgdbCandidates({ clientId, token, cache, total: CANDIDATES });
  const { kept, excluded } = selectGames(raws, config, POOL_SIZE);

  const unmapped = new Map();
  const onUnmapped = name => unmapped.set(name, (unmapped.get(name) || 0) + 1);
  const games = [];
  let withSteamTags = 0;

  for (const [i, raw] of kept.entries()) {
    const appId = steamAppId(raw);
    let steam = { tags: null, metacritic: null };
    if (appId) {
      steam = await fetchSteamData(appId, { cache });
      if (steam.failed) console.warn(`  Steam fetch failed for ${raw.name} (app ${appId}); using IGDB fallback`);
    }
    const entry = buildEntry(raw, steam, config, onUnmapped);
    if (!entry.primaryTags.length) { excluded.push({ id: raw.id, name: raw.name, reason: 'no tags' }); continue; }
    if (steam.tags && steam.tags.length) withSteamTags++;
    games.push(entry);
    if ((i + 1) % 100 === 0) console.log(`Enriched ${i + 1}/${kept.length}`);
  }

  games.sort((a, b) => a.name.localeCompare(b.name) || a.year - b.year);
  const today = new Date().toISOString().slice(0, 10);
  fs.writeFileSync(path.join(ROOT, 'data.js'), renderDataJs(games, today));
  fs.writeFileSync(path.join(ROOT, 'excluded.txt'), renderExcluded(excluded));

  const byReason = {};
  for (const e of excluded) {
    const reason = e.reason.split(':')[0].replace(/ of \d+$/, '');
    byReason[reason] = (byReason[reason] || 0) + 1;
  }
  console.log('\n── Summary ──');
  console.log(`Kept: ${games.length}`);
  console.log(`Excluded: ${excluded.length}`, byReason);
  console.log(`With Steam tags: ${pct(withSteamTags, games.length)}`);
  console.log(`Null metacritic: ${pct(games.filter(g => g.metacritic == null).length, games.length)}`);
  if (unmapped.size) console.log('Unmapped platforms (add to platformMap?):', Object.fromEntries(unmapped));
}

if (require.main === module) {
  main().catch(err => { console.error(err.message); process.exit(1); });
}

module.exports = { renderDataJs, renderExcluded };
```

```js
'use strict';

// ═══════════════════════════════════════════════════════════════════════════
//  STATE
// ═══════════════════════════════════════════════════════════════════════════
const state = {
  mode: 'daily',
  dateStr: todayStr(),
  answer: null,
  guesses: [],
  suggestions: [],
  highlighted: -1,
  lastAddedId: null,
};

const $ = id => document.getElementById(id);

function el(tag, className, text) {
  const node = document.createElement(tag);
  if (className) node.className = className;
  if (text != null) node.textContent = text;
  return node;
}

function storageKey(dateStr) { return `gtgdle_${dateStr}`; }
function currentStatus() { return gameStatus(state.guesses, state.answer); }

// ═══════════════════════════════════════════════════════════════════════════
//  GAME FLOW
// ═══════════════════════════════════════════════════════════════════════════
function startDaily() {
  state.mode = 'daily';
  state.dateStr = todayStr();
  state.answer = pickDaily(GAMES, state.dateStr);
  let saved = null;
  try { saved = parseSavedState(localStorage.getItem(storageKey(state.dateStr)), GAMES); } catch { saved = null; }
  state.guesses = saved ? saved.guesses : [];
  state.lastAddedId = null;
  render();
}

function startFree() {
  state.mode = 'free';
  state.answer = pickRandom(GAMES);
  state.guesses = [];
  state.lastAddedId = null;
  render();
}

function submitGuess(game) {
  if (currentStatus() !== 'playing') return;
  if (state.guesses.some(g => g.id === game.id)) { flash('Already guessed'); return; }
  state.guesses.push(game);
  state.lastAddedId = game.id;
  if (state.mode === 'daily') {
    try { localStorage.setItem(storageKey(state.dateStr), serializeState(state.guesses)); } catch { /* storage full/blocked — play on */ }
  }
  $('guess-input').value = '';
  closeSuggestions();
  render();
}

function flash(message) {
  $('flash').textContent = message;
  clearTimeout(flash.timer);
  flash.timer = setTimeout(() => { $('flash').textContent = ''; }, 2000);
}

// ═══════════════════════════════════════════════════════════════════════════
//  RENDER
// ═══════════════════════════════════════════════════════════════════════════
function render() {
  renderHeader();
  renderPips();
  renderRows();
  const playing = currentStatus() === 'playing';
  $('guess-input').disabled = !playing;
  $('guess-input').placeholder = playing ? `Guess ${state.guesses.length + 1} of ${MAX_GUESSES}…` : 'Game over';
}

function renderHeader() {
  $('btn-daily').classList.toggle('active', state.mode === 'daily');
  $('btn-free').classList.toggle('active', state.mode === 'free');
  $('day-label').textContent = state.mode === 'daily'
    ? `#${dayNumber(state.dateStr)} · ${new Date(`${state.dateStr}T12:00:00`).toLocaleDateString('en-US', { month: 'long', day: 'numeric', year: 'numeric' })}`
    : 'Free Play — random game';
}

function renderPips() {
  const pips = $('pips');
  pips.replaceChildren();
  const won = currentStatus() === 'won';
  for (let i = 0; i < MAX_GUESSES; i++) {
    const pip = el('span', 'pip');
    if (i < state.guesses.length) pip.classList.add(won && i === state.guesses.length - 1 ? 'win' : 'used');
    pips.appendChild(pip);
  }
}

function renderRows() {
  const rows = $('rows');
  rows.replaceChildren();
  [...state.guesses].reverse().forEach(g => rows.appendChild(renderRow(g, compareGuess(g, state.answer))));
}

function renderRow(game, result) {
  const row = el('div', 'row');
  if (game.id === state.lastAddedId) row.classList.add('new');

  const name = el('div', 'row-name', `${game.name} `);
  name.appendChild(el('span', 'year', `(${game.year})`));
  if (game.id === state.answer.id) name.classList.add('correct');
  row.appendChild(name);

  const tiles = el('div', 'tiles');
  COLUMNS.forEach((col, i) => {
    const cell = result[col];
    const tile = el('div', `tile ${cell.status}`);
    tile.style.setProperty('--i', i);
    tile.appendChild(el('span', 'tile-value', cell.display));
    if (cell.arrow) tile.appendChild(el('span', 'tile-arrow', cell.arrow === 'up' ? '↑' : '↓'));
    if (col === 'tags' && cell.matched && cell.matched.length > TAG_DISPLAY_LIMIT) tile.title = cell.matched.join(', ');
    tiles.appendChild(tile);
  });
  row.appendChild(tiles);
  return row;
}

// ═══════════════════════════════════════════════════════════════════════════
//  AUTOCOMPLETE
// ═══════════════════════════════════════════════════════════════════════════
function updateSuggestions() {
  const guessedIds = new Set(state.guesses.map(g => g.id));
  state.suggestions = searchGames(GAMES, $('guess-input').value, guessedIds);
  state.highlighted = state.suggestions.length ? 0 : -1;
  renderSuggestions();
}

function renderSuggestions() {
  const list = $('suggestions');
  list.replaceChildren();
  state.suggestions.forEach((g, i) => {
    const li = el('li', i === state.highlighted ? 'highlighted' : '');
    li.setAttribute('role', 'option');
    li.appendChild(el('span', 'name', g.name));
    li.appendChild(el('span', 'year', String(g.year)));
    li.addEventListener('mousedown', e => { e.preventDefault(); submitGuess(g); });
    list.appendChild(li);
  });
  list.hidden = state.suggestions.length === 0;
}

function closeSuggestions() {
  state.suggestions = [];
  state.highlighted = -1;
  renderSuggestions();
}

function onInputKey(e) {
  const n = state.suggestions.length;
  if (e.key === 'ArrowDown' && n) { e.preventDefault(); state.highlighted = (state.highlighted + 1) % n; renderSuggestions(); }
  else if (e.key === 'ArrowUp' && n) { e.preventDefault(); state.highlighted = (state.highlighted - 1 + n) % n; renderSuggestions(); }
  else if (e.key === 'Enter') {
    e.preventDefault();
    if (state.highlighted >= 0) submitGuess(state.suggestions[state.highlighted]);
    else if ($('guess-input').value.trim()) flash('Pick a game from the list');
  }
  else if (e.key === 'Escape') closeSuggestions();
}

// ═══════════════════════════════════════════════════════════════════════════
//  BOOT
// ═══════════════════════════════════════════════════════════════════════════
function boot() {
  if (typeof GAMES === 'undefined' || !Array.isArray(GAMES) || GAMES.length === 0) {
    $('error-banner').hidden = false;
    $('search').hidden = true;
    return;
  }
  $('btn-daily').addEventListener('click', startDaily);
  $('btn-free').addEventListener('click', startFree);
  $('guess-input').addEventListener('input', updateSuggestions);
  $('guess-input').addEventListener('keydown', onInputKey);
  $('guess-input').addEventListener('blur', closeSuggestions);
  startDaily();
}

boot();
```

```js
// ═══════════════════════════════════════════════════════════════════════════
//  RESULT
// ═══════════════════════════════════════════════════════════════════════════
let resultDismissed = false;

function renderResult() {
  const status = currentStatus();
  const modal = $('result-modal');
  if (status === 'playing' || resultDismissed) { modal.hidden = true; return; }

  $('result-title').textContent = status === 'won'
    ? `You got it in ${state.guesses.length}!`
    : 'Out of guesses';
  const answer = $('result-answer');
  answer.replaceChildren('The answer was ', el('strong', '', state.answer.name), ` (${state.answer.year}).`);
  $('btn-new').hidden = state.mode !== 'free';
  $('share-fallback').hidden = true;
  $('share-msg').textContent = '';
  modal.hidden = false;
}

function shareResult() {
  const text = buildShareText({
    mode: state.mode,
    dateStr: state.dateStr,
    results: state.guesses.map(g => compareGuess(g, state.answer)),
    won: currentStatus() === 'won',
  });
  const showFallback = () => {
    $('share-fallback').textContent = text;
    $('share-fallback').hidden = false;
    $('share-msg').textContent = 'Copy the text above';
  };
  if (navigator.clipboard && navigator.clipboard.writeText) {
    navigator.clipboard.writeText(text).then(() => { $('share-msg').textContent = 'Copied to clipboard!'; }, showFallback);
  } else {
    showFallback();
  }
}
```
